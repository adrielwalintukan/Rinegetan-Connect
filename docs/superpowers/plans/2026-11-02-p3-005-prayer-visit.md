## Global Constraints

- Batas panjang menghitung Unicode code point; teks dipangkas (`trim`, `\r\n` → `\n`) sebelum divalidasi.
- Doa: nama 0-100, kontak 0-254 (email sederhana atau telepon 8-15 digit bila diisi), isi doa 1-2000, `wantsContact` hanya benar bila kontak diisi.
- Kunjungan: nama 1-100, kontak 1-254, `visitDate` `YYYY-MM-DD` valid dan tidak sebelum hari ini di Asia/Makassar (opsional), `partySize` bilangan bulat 1-20 (opsional), pesan 0-1000.
- Consent version kedua form: `draft-2026-11-02`. Copy retensi: "Data disimpan paling lama 24 bulan, lalu dihapus."
- Rate limit per form: `<form>_attempt` 20 per jam sebelum Turnstile, `<form>` 5 per jam setelah Turnstile; form key `prayer` dan `visit`.
- Turnstile fail-closed (tanpa secret → 503, gagal/timeout 5 detik → 400). Body JSON wajib (415), maksimal 8 KB (413).
- Tidak ada email, tidak ada logging isi, audit hanya status. Tanpa dependency baru.
- Route dan perilaku P3-004 (`/api/minat-belajar`, `/api/staff/study-interest/[id]`) tidak berubah; seluruh tes P3-004 tetap lulus tanpa diubah.
- Route tidak boleh meneruskan `error.message` mentah (tes p2-502); gunakan `publicErrorMessage`.
- Commit tanpa atribusi AI, diakhiri `Refs #48`.

// frontend/src/lib/forms/public-form.mjs
import { createHmac } from "node:crypto";
import { UserFacingError, GENERIC_ERROR_MESSAGE } from "../api/errors.mjs";

export const MAX_FORM_BYTES = 8192;
const EMAIL = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
const PHONE = /^[0-9 +()\-]+$/;

export class PublicFormError extends UserFacingError {
  /** @param {string} message @param {number} [status] @param {string} [field] */
  constructor(message, status = 400, field) { super(message); this.name = "PublicFormError"; this.status = status; this.field = field; }
}
/** @param {unknown} value @param {string} field @param {string} label @param {number} max @param {boolean} required */
export function formText(value, field, label, max, required) {
  if (value === undefined && !required) return "";
  if (typeof value !== "string") throw new PublicFormError(`${label} tidak valid.`, 400, field);
  const clean = value.replace(/\r\n?/g, "\n").trim();
  if (clean.includes("\0") || !clean.isWellFormed()) throw new PublicFormError(`${label} berisi karakter yang tidak diizinkan.`, 400, field);
  if (required && !clean) throw new PublicFormError(`${label} wajib diisi.`, 400, field);
  if (Array.from(clean).length > max) throw new PublicFormError(`${label} maksimal ${max.toLocaleString("id-ID")} karakter.`, 400, field);
  return clean;
}
/** @param {string} value */
export function isContact(value) {
  const digits = value.replace(/\D/g, "").length;
  return EMAIL.test(value) || (PHONE.test(value) && digits >= 8 && digits <= 15);
}
/** Checks shape and allow-listed keys; returns null when the honeypot is filled.
 * @param {unknown} input @param {string[]} keys @returns {Record<string, unknown> | null} */
export function formFields(input, keys) {
  if (!input || typeof input !== "object" || Array.isArray(input)) throw new PublicFormError("Format permintaan tidak valid.");
  const data = /** @type {Record<string, unknown>} */ (input);
  if (Object.keys(data).some((key) => !keys.includes(key))) throw new PublicFormError("Format permintaan tidak valid.");
  // Bots that fill the honeypot get the generic success before any field-level feedback.
  if (typeof data.website === "string" && data.website.trim() !== "") return null;
  return data;
}
/** @param {Record<string, unknown>} data */
export function turnstileToken(data) { return typeof data.turnstileToken === "string" ? data.turnstileToken : ""; }

/** @param {string} ip @param {string} key */
export function hashFormIp(ip, key) { return createHmac("sha256", key).update(ip).digest("hex").slice(0, 32); }
/** @param {Request} request */
export function clientIp(request) { return request.headers.get("x-forwarded-for")?.split(",")[0]?.trim() || "127.0.0.1"; }

/** @param {Request} request @returns {Promise<unknown>} */
export async function readFormBody(request) {
  if (!(request.headers.get("content-type") || "").toLowerCase().startsWith("application/json")) throw new PublicFormError("Format permintaan tidak valid.", 415);
  const tooBig = () => new PublicFormError("Isian terlalu panjang.", 413);
  if (Number(request.headers.get("content-length")) > MAX_FORM_BYTES) throw tooBig();
  const chunks = []; let total = 0;
  if (request.body) {
    const reader = request.body.getReader();
    for (;;) {
      const { done, value } = await reader.read();
      if (done) break;
      if (total + value.length > MAX_FORM_BYTES) { await reader.cancel(); throw tooBig(); }
      chunks.push(value); total += value.length;
    }
  }
  const bytes = new Uint8Array(total); let offset = 0;
  for (const chunk of chunks) { bytes.set(chunk, offset); offset += chunk.length; }
  try { return JSON.parse(new TextDecoder("utf-8", { fatal: true }).decode(bytes)); }
  catch { throw new PublicFormError("Format permintaan tidak valid.", 400); }
}

/** @param {string} token @param {string} secret @param {typeof fetch} [fetchImpl] */
export async function verifyTurnstile(token, secret, fetchImpl = fetch) {
  try {
    const res = await fetchImpl("https://challenges.cloudflare.com/turnstile/v0/siteverify", {
      method: "POST", body: new URLSearchParams({ secret, response: token }), signal: AbortSignal.timeout(5000),
    });
    const json = await res.json();
    return json?.success === true;
  } catch { return false; }
}

/**
 * @param {unknown} input
 * @param {{ key: string, success: string, parse: (input: unknown) => { honeypot: boolean, turnstileToken?: string }, toRow: (parsed: any) => object }} form
 * @param {{ ipHash: string, turnstileSecret?: string, verifyTurnstile: (token: string, secret: string) => Promise<boolean>, checkRate: (form: string, ipHash: string, limit: number) => Promise<boolean>, insertRow: (row: object) => Promise<void> }} deps
 */
export async function submitPublicForm(input, form, deps) {
  let parsed;
  try { parsed = form.parse(input); }
  catch (error) {
    if (error instanceof PublicFormError) return { status: error.status, body: { error: error.message, field: error.field } };
    throw error;
  }
  if (parsed.honeypot) return { status: 200, body: { success: true, message: form.success } };
  const limited = { status: 429, body: { error: "Terlalu banyak permintaan. Silakan coba lagi nanti." } };
  try {
    if (!(await deps.checkRate(`${form.key}_attempt`, deps.ipHash, 20))) return limited;
    if (!deps.turnstileSecret) return { status: 503, body: { error: "Formulir sementara tidak tersedia." } };
    if (!parsed.turnstileToken || !(await deps.verifyTurnstile(parsed.turnstileToken, deps.turnstileSecret))) {
      return { status: 400, body: { error: "Verifikasi gagal, silakan coba lagi.", field: "turnstile" } };
    }
    if (!(await deps.checkRate(form.key, deps.ipHash, 5))) return limited;
    await deps.insertRow(form.toRow(parsed));
  } catch { return { status: 500, body: { error: GENERIC_ERROR_MESSAGE } }; }
  return { status: 201, body: { success: true, message: form.success } };
}

import { PublicFormError, formFields, formText, isContact, submitPublicForm, turnstileToken } from "./public-form.mjs";

export { PublicFormError as StudyInterestError, MAX_FORM_BYTES as MAX_STUDY_INTEREST_BYTES, hashFormIp, clientIp, readFormBody, verifyTurnstile } from "./public-form.mjs";
export const STUDY_INTEREST_CONSENT_VERSION = "2026-10-29";
const KEYS = ["name", "contact", "message", "consent", "website", "turnstileToken"];
const SUCCESS = "Terima kasih. Tim kami akan menghubungi Anda.";

/** @param {unknown} input */
export function parseStudyInterest(input) {
  const data = formFields(input, KEYS);
  if (!data) return { name: "", contact: "", message: "", honeypot: true, turnstileToken: "" };
  const name = formText(data.name, "name", "Nama", 100, true);
  const contact = formText(data.contact, "contact", "Kontak", 254, true);
  if (!isContact(contact)) throw new PublicFormError("Isi nomor WhatsApp atau email yang valid.", 400, "contact");
  const message = formText(data.message, "message", "Pesan", 1000, false);
  if (data.consent !== true) throw new PublicFormError("Centang persetujuan untuk melanjutkan.", 400, "consent");
  return { name, contact, message, honeypot: false, turnstileToken: turnstileToken(data) };
}

const FORM = {
  key: "study_interest", success: SUCCESS, parse: parseStudyInterest,

};

as $$
begin
  if tg_op = 'INSERT' then

    new.handled_by := null;
    new.handled_at := null;
    new.created_at := now();
    new.consented_at := now();
    return new;
  end if;

  if (to_jsonb(new) - 'handled_by') = (to_jsonb(old) - 'handled_by')
    and new.handled_by is null and old.handled_by is not null
    and not exists (select 1 from auth.users where id = old.handled_by)

    return new;

  const data = formFields(input, KEYS);

  if (value === undefined || value === "") return null;
  const date = typeof value === "string" && DATE.test(value) ? new Date(`${value}T00:00:00Z`) : null;
  if (!date || Number.isNaN(date.getTime()) || !date.toISOString().startsWith(String(value))) throw new PublicFormError("Tanggal kunjungan tidak valid.", 400, "visitDate");
  if (String(value) < makassarToday(now)) throw new PublicFormError("Tanggal kunjungan tidak boleh sudah lewat.", 400, "visitDate");
  return String(value);
}
/** @param {unknown} value */
function partySize(value) {
  if (value === undefined || value === "") return null;
  if (typeof value !== "number" || !Number.isInteger(value) || value < 1 || value > 20) throw new PublicFormError("Jumlah orang antara 1 dan 20.", 400, "partySize");
  return value;
}

/** @param {unknown} input @param {Date} [now] */
export function parseVisit(input, now = new Date()) {
  const data = formFields(input, KEYS);
  if (!data) return { name: "", contact: "", visitDate: null, partySize: null, message: "", honeypot: true, turnstileToken: "" };
  const name = formText(data.name, "name", "Nama", 100, true);
  const contact = formText(data.contact, "contact", "Kontak", 254, true);
  if (!isContact(contact)) throw new PublicFormError("Isi nomor WhatsApp atau email yang valid.", 400, "contact");
  const date = visitDate(data.visitDate, now);
  const size = partySize(data.partySize);
  const message = formText(data.message, "message", "Pesan", 1000, false);
  if (data.consent !== true) throw new PublicFormError("Centang persetujuan untuk melanjutkan.", 400, "consent");
  return { name, contact, visitDate: date, partySize: size, message, honeypot: false, turnstileToken: turnstileToken(data) };
}

/** @param {unknown} input @param {Parameters<typeof submitPublicForm>[2] & { now?: Date }} deps */
export function submitVisit(input, deps) {

  visitor_submissions: { columns: "id, name, contact, visit_date, party_size, message, status, created_at, handled_at", statuses: ["new", "contacted", "closed"] },

/** 50 newest rows; RLS already limits rows to active Admins.
 * @param {import('@supabase/supabase-js').SupabaseClient} client @param {keyof typeof PASTORAL_TABLES} table */
export async function loadPastoralRows(client, table) {
  try {
    const { data, error } = await client.from(table).select(PASTORAL_TABLES[table].columns).order("created_at", { ascending: false }).limit(50);
    return error || !data ? { data: null, error: true } : { data, error: false };
  } catch { return { data: null, error: true }; }
}

/** @param {unknown} input @param {keyof typeof PASTORAL_TABLES} table @returns {string} */
export function parsePastoralStatus(input, table) {
  const data = input && typeof input === "object" && !Array.isArray(input) ? /** @type {Record<string, unknown>} */ (input) : null;
  if (!data || Object.keys(data).some((key) => key !== "status") || typeof data.status !== "string" || !PASTORAL_TABLES[table].statuses.includes(data.status)) {
    throw new PublicFormError("Status tidak valid.", 400);
  }
  return data.status;
}

/** @param {import('@supabase/supabase-js').SupabaseClient} client @param {keyof typeof PASTORAL_TABLES} table
 * @param {string} id @param {string} status @param {'admin' | 'editor'} role */
export async function updatePastoralStatus(client, table, id, status, role) {
  if (role !== "admin") throw new PublicFormError("Hanya Admin yang dapat mengelola data pastoral.", 403);
  if (typeof id !== "string" || !uuid.test(id)) throw new PublicFormError("Data tidak valid.", 400);
  const { data, error } = await client.from(table).update({ status }).eq("id", id).select("id, status").maybeSingle();
  if (error) throw new PublicFormError(publicErrorMessage(error), error.code === "42501" ? 403 : 500);
  if (!data) throw new PublicFormError("Data tidak ditemukan.", 404);
  return data;
}

// frontend/src/components/forms/useTurnstile.ts
"use client";

import { useCallback, useEffect, useRef, useState } from "react";

type Turnstile = {
  render: (element: HTMLElement, options: Record<string, unknown>) => string;
  reset: (widgetId?: string) => void;
};
const turnstile = () => (window as Window & { turnstile?: Turnstile }).turnstile;

export const TURNSTILE_SITE_KEY = process.env.NEXT_PUBLIC_TURNSTILE_SITE_KEY;
export const TURNSTILE_SCRIPT = "https://challenges.cloudflare.com/turnstile/v0/api.js?render=explicit";

// One explicit widget per form; render runs on mount and again when the script reports ready.
export function useTurnstile() {
  const [token, setToken] = useState("");
  const ref = useRef<HTMLDivElement>(null);
  const widgetId = useRef<string | undefined>(undefined);
  const render = useCallback(() => {
    const api = turnstile();
    if (!TURNSTILE_SITE_KEY || !ref.current || !api || widgetId.current !== undefined) return;
    widgetId.current = api.render(ref.current, {
      sitekey: TURNSTILE_SITE_KEY, language: "id", size: "flexible", callback: setToken, "expired-callback": () => setToken(""), "error-callback": () => setToken(""),
    });
  }, []);
  useEffect(render, [render]);
  const reset = useCallback(() => { setToken(""); turnstile()?.reset(widgetId.current); }, []);
  return { token, ref, render, reset };
}
